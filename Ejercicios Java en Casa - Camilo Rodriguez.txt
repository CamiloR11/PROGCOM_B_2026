5. Cuadratica
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Ingrese el valor de a: ");
        double a = sc.nextDouble();

        System.out.print("Ingrese el valor de b: ");
        double b = sc.nextDouble();

        System.out.print("Ingrese el valor de c: ");
        double c = sc.nextDouble();

        double d = b * b - 4 * a * c;

        if (d > 0) {
            System.out.println("La ecuación tiene dos soluciones reales");
        } else if (d == 0) {
            System.out.println("La ecuación tiene una solución real");
        } else {
            System.out.println("La ecuación no tiene soluciones reales");
        }
    }
}

10.Tipos de Triangulo
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Ingrese el primer lado: ");
        int a = sc.nextInt();

        System.out.print("Ingrese el segundo lado: ");
        int b = sc.nextInt();

        System.out.print("Ingrese el tercer lado: ");
        int c = sc.nextInt();

        if (a == b && b == c) {
            System.out.println("El triángulo es equilátero");
        } else if (a == b || a == c || b == c) {
            System.out.println("El triángulo es isósceles");
        } else {
            System.out.println("El triángulo es escaleno");
        }
    }
}

11. IMC
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Ingrese su nombre: ");
        String nombre = sc.next();

        System.out.print("Ingrese su peso en kg: ");
        double peso = sc.nextDouble();

        System.out.print("Ingrese su altura en metros: ");
        double altura = sc.nextDouble();

        double imc = peso / (altura * altura);

        if (imc < 18.5) {
            System.out.println(nombre + ", su IMC indica bajo peso");
        } else if (imc < 25) {
            System.out.println(nombre + ", su IMC es normal");
        } else if (imc < 30) {
            System.out.println(nombre + ", su IMC indica sobrepeso");
        } else {
            System.out.println(nombre + ", su IMC indica obesidad");
        }
    }
}

12. Calculadora Simple
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Ingrese el primer número: ");
        double a = sc.nextDouble();

        System.out.print("Ingrese el segundo número: ");
        double b = sc.nextDouble();

        System.out.print("Ingrese la operación (+ - * /): ");
        char op = sc.next().charAt(0);

        if (op == '+') System.out.println("Resultado: " + (a + b));
        else if (op == '-') System.out.println("Resultado: " + (a - b));
        else if (op == '*') System.out.println("Resultado: " + (a * b));
        else if (op == '/') System.out.println("Resultado: " + (a / b));
    }
}

15. Formar Triangulo
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Ingrese el primer lado: ");
        int a = sc.nextInt();

        System.out.print("Ingrese el segundo lado: ");
        int b = sc.nextInt();

        System.out.print("Ingrese el tercer lado: ");
        int c = sc.nextInt();

        if (a + b > c && a + c > b && b + c > a) {
            if (a == b && b == c) {
                System.out.println("El triángulo es equilátero");
            } else if (a == b || a == c || b == c) {
                System.out.println("El triángulo es isósceles");
            } else {
                System.out.println("El triángulo es escaleno");
            }
        } else {
            System.out.println("Los valores no forman un triángulo");
        }
    }
}
